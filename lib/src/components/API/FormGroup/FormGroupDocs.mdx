import { useState } from "react";

function FruitOptions() {
const [values, setValues] = useState({});

const handleChange = (e) => {
const { name, checked } = e.target;
setValues((prev) => ({ ...prev, [name]: checked }));
};

return (

<FormGroup row>
  <FormControlLabel
    control={<Checkbox name="apple" onChange={handleChange} />}
    label="Apple"
  />
  <FormControlLabel
    control={<Checkbox name="banana" onChange={handleChange} />}
    label="Banana"
  />
  <FormControlLabel
    control={<Checkbox name="cherry" onChange={handleChange} />}
    label="Cherry"
  />
</FormGroup>
); } `} />
