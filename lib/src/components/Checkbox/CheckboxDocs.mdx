import { Checkbox } from "@base-xui/react-xui";

export default function Home() {
  const [checked, setChecked] = useState(false);
  return (
    <div>
      <Checkbox
        id="checkbox-id"
        name="checkbox-name"
        className=""
        color="primary"
        size="medium"
        required={false}
        disabled={false}
        indeterminate={false}
        checked={checked}
        value={"checkbox-value"}
        onChange={(e: ChangeEvent<HTMLInputElement>) =>
          setChecked(e.target.checked)
        }
        checkedIcon={<Check />}
      />
    </div>
  );
}
